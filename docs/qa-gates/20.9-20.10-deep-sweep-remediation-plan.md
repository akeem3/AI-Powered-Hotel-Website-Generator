### Issue #9: No Atomic Rollback Mechanism

**Current State:**
- Script doesn't implement atomic file writes
- No rollback on failure

**Required Enhancement:**
If implementing full AST manipulation (Issue #5 Option A):
```typescript
async function updateFilesAtomically(updates: FileUpdate[]) {
  const backups = new Map<string, string>();

  try {
    // Create backups
    for (const update of updates) {
      backups.set(update.path, fs.readFileSync(update.path, 'utf-8'));
    }

    // Apply updates
    for (const update of updates) {
      fs.writeFileSync(update.path, update.content, 'utf-8');
    }

  } catch (error) {
    // Rollback
    for (const [path, content] of backups) {
      fs.writeFileSync(path, content, 'utf-8');
    }
    throw error;
  }
}
```
